    <script>
        // ==================== DATA STORE ====================
        let employees = [];
        let attendance = {};
        let payrollData = {};
        let employeeIdCounter = 1;

        // ==================== HELPER: Format Indian Rupee ====================
        function formatINR(amount) {
            return '₹' + Number(amount).toLocaleString('en-IN');
        }

        // ==================== INITIAL DATA (in INR) ====================
        function initData() {
            employees = [{
                id: 1,
                name: "Jessica Diaz",
                email: "jessica@company.com",
                department: "Engineering",
                designation: "Senior Developer",
                status: "Active",
                salary: 62000
            }, {
                id: 2,
                name: "Ryan Chen",
                email: "ryan@company.com",
                department: "Engineering",
                designation: "Team Lead",
                status: "Active",
                salary: 78000
            }, {
                id: 3,
                name: "Maya Lopez",
                email: "maya@company.com",
                department: "Marketing",
                designation: "Manager",
                status: "On Leave",
                salary: 54000
            }, {
                id: 4,
                name: "Tyler White",
                email: "tyler@company.com",
                department: "Finance",
                designation: "Analyst",
                status: "Active",
                salary: 48000
            }, {
                id: 5,
                name: "Priya Nair",
                email: "priya@company.com",
                department: "HR",
                designation: "Manager",
                status: "Active",
                salary: 59500
            }];
            employeeIdCounter = 6;

            // Init attendance
            employees.forEach(emp => {
                attendance[emp.id] = {
                    present: Math.floor(Math.random() * 20) + 5,
                    total: 22,
                    today: Math.random() > 0.3 ? 'Present' : 'Absent'
                };
            });

            // Init payroll (in INR)
            employees.forEach(emp => {
                const salary = emp.salary;
                const basic = salary * 0.5;
                const hra = salary * 0.2;
                const da = salary * 0.1;
                const bonus = salary * 0.05;
                const deductions = salary * 0.12;
                payrollData[emp.id] = {
                    basic: Math.round(basic),
                    hra: Math.round(hra),
                    da: Math.round(da),
                    bonus: Math.round(bonus),
                    deductions: Math.round(deductions),
                    net: Math.round(basic + hra + da + bonus - deductions)
                };
            });

            renderAll();
        }

        // ==================== RENDER FUNCTIONS ====================
        function renderAll() {
            renderStats();
            renderDashboardTable();
            renderEmployeeTable();
            renderAttendanceTable();
            renderPayrollTable();
        }

        function renderStats() {
            document.getElementById('totalEmployees').textContent = employees.length;
            const total = employees.reduce((sum, e) => sum + e.salary, 0);
            document.getElementById('totalSalary').textContent = formatINR(total);

            const present = Object.values(attendance).filter(a => a.today === 'Present').length;
            document.getElementById('presentToday').textContent = present;

            const leave = employees.filter(e => e.status === 'On Leave').length;
            document.getElementById('onLeave').textContent = leave;
        }

        function renderDashboardTable() {
            const tbody = document.getElementById('dashboardTableBody');
            if (employees.length === 0) {
                tbody.innerHTML = `<tr><td colspan="4" class="empty-state">No employees yet</td></tr>`;
                return;
            }
            tbody.innerHTML = employees.slice(0, 5).map(emp => `
                <tr>
                    <td><span class="avatar">${emp.name.split(' ').map(n=>n[0]).join('')}</span> ${emp.name}</td>
                    <td>${emp.department}</td>
                    <td><span class="badge ${emp.status.toLowerCase().replace(' ', '')}">${emp.status}</span></td>
                    <td><strong>${formatINR(emp.salary)}</strong></td>
                </tr>
            `).join('');
        }

        function renderEmployeeTable() {
            const tbody = document.getElementById('employeeTableBody');
            if (employees.length === 0) {
                tbody.innerHTML = `<tr><td colspan="8" class="empty-state"><i class="fas fa-users"></i><br/>No employees registered</td></tr>`;
                return;
            }
            tbody.innerHTML = employees.map((emp, idx) => `
                <tr>
                    <td>${idx + 1}</td>
                    <td><span class="avatar">${emp.name.split(' ').map(n=>n[0]).join('')}</span> ${emp.name}</td>
                    <td contenteditable="true" onblur="updateField(${emp.id}, 'email', this.textContent)">${emp.email}</td>
                    <td contenteditable="true" onblur="updateField(${emp.id}, 'department', this.textContent)">${emp.department}</td>
                    <td contenteditable="true" onblur="updateField(${emp.id}, 'designation', this.textContent)">${emp.designation}</td>
                    <td>
                        <select onchange="updateField(${emp.id}, 'status', this.value)" style="padding:4px 8px;border-radius:8px;border:1px solid #d4dfed;background:white;">
                            <option value="Active" ${emp.status==='Active'?'selected':''}>Active</option>
                            <option value="Inactive" ${emp.status==='Inactive'?'selected':''}>Inactive</option>
                            <option value="On Leave" ${emp.status==='On Leave'?'selected':''}>On Leave</option>
                        </select>
                    </td>
                    <td contenteditable="true" onblur="updateSalary(${emp.id}, this.textContent)">${formatINR(emp.salary)}</td>
                    <td class="action-icons">
                        <i class="fas fa-edit" onclick="editEmployee(${emp.id})" title="Edit"></i>
                        <i class="fas fa-trash" onclick="deleteEmployee(${emp.id})" style="color:#c0392b;" title="Delete"></i>
                    </td>
                </tr>
            `).join('');
        }

        function renderAttendanceTable() {
            const tbody = document.getElementById('attendanceTableBody');
            if (employees.length === 0) {
                tbody.innerHTML = `<tr><td colspan="4" class="empty-state">No attendance records</td></tr>`;
                return;
            }
            tbody.innerHTML = employees.map(emp => {
                const att = attendance[emp.id] || { present: 0, total: 22, today: 'Absent' };
                return `
                    <tr>
                        <td><span class="avatar">${emp.name.split(' ').map(n=>n[0]).join('')}</span> ${emp.name}</td>
                        <td><span class="badge ${att.today === 'Present' ? 'active' : 'inactive'}">${att.today}</span></td>
                        <td>${att.present} / ${att.total}</td>
                        <td><span class="badge ${emp.status.toLowerCase().replace(' ', '')}">${emp.status}</span></td>
                    </tr>
                `;
            }).join('');
        }

        function renderPayrollTable() {
            const tbody = document.getElementById('payrollTableBody');
            if (employees.length === 0) {
                tbody.innerHTML = `<tr><td colspan="7" class="empty-state">No payroll data</td></tr>`;
                return;
            }
            let gross = 0,
                net = 0;
            tbody.innerHTML = employees.map(emp => {
                const p = payrollData[emp.id] || { basic: 0, hra: 0, da: 0, bonus: 0, deductions: 0, net: 0 };
                gross += emp.salary;
                net += p.net;
                return `
                    <tr>
                        <td><span class="avatar">${emp.name.split(' ').map(n=>n[0]).join('')}</span> ${emp.name}</td>
                        <td>${formatINR(p.basic)}</td>
                        <td>${formatINR(p.hra)}</td>
                        <td>${formatINR(p.da)}</td>
                        <td>${formatINR(p.bonus)}</td>
                        <td>${formatINR(p.deductions)}</td>
                        <td><strong>${formatINR(p.net)}</strong></td>
                    </tr>
                `;
            }).join('');
            document.getElementById('grossPayroll').textContent = formatINR(gross);
            document.getElementById('netPayroll').textContent = formatINR(net);
            document.getElementById('payrollCount').textContent = employees.length;
        }

        // ==================== CRUD OPERATIONS ====================
        function updateField(id, field, value) {
            const emp = employees.find(e => e.id === id);
            if (emp) {
                emp[field] = value.trim();
                renderAll();
                showToast('Updated ' + field);
            }
        }

        function updateSalary(id, value) {
            const emp = employees.find(e => e.id === id);
            if (emp) {
                const num = parseInt(value.replace(/[₹,]/g, ''));
                if (!isNaN(num) && num > 0) {
                    emp.salary = num;
                    // Update payroll
                    const basic = emp.salary * 0.5;
                    const hra = emp.salary * 0.2;
                    const da = emp.salary * 0.1;
                    const bonus = emp.salary * 0.05;
                    const deductions = emp.salary * 0.12;
                    payrollData[emp.id] = {
                        basic: Math.round(basic),
                        hra: Math.round(hra),
                        da: Math.round(da),
                        bonus: Math.round(bonus),
                        deductions: Math.round(deductions),
                        net: Math.round(basic + hra + da + bonus - deductions)
                    };
                    renderAll();
                    showToast('Salary updated');
                }
            }
        }

        function deleteEmployee(id) {
            if (confirm('Delete this employee?')) {
                employees = employees.filter(e => e.id !== id);
                delete attendance[id];
                delete payrollData[id];
                renderAll();
                showToast('Employee deleted');
            }
        }

        function editEmployee(id) {
            const emp = employees.find(e => e.id === id);
            if (!emp) return;
            const modal = document.getElementById('modal');
            const content = document.getElementById('modalContent');
            content.innerHTML = `
                <h2><i class="fas fa-edit"></i> Edit Employee</h2>
                <div class="form-group">
                    <label>Name</label>
                    <input type="text" id="editName" value="${emp.name}" />
                </div>
                <div class="form-group">
                    <label>Email</label>
                    <input type="email" id="editEmail" value="${emp.email}" />
                </div>
                <div class="form-row" style="display:grid;grid-template-columns:1fr 1fr;gap:1rem;">
                    <div class="form-group">
                        <label>Department</label>
                        <input type="text" id="editDept" value="${emp.department}" />
                    </div>
                    <div class="form-group">
                        <label>Designation</label>
                        <input type="text" id="editDesig" value="${emp.designation}" />
                    </div>
                </div>
                <div class="form-group">
                    <label>Salary (₹)</label>
                    <input type="number" id="editSalary" value="${emp.salary}" />
                </div>
                <div class="modal-actions">
                    <button class="btn" onclick="closeModal()">Cancel</button>
                    <button class="btn btn-primary" onclick="saveEdit(${id})">Save</button>

        }

        function saveEdit(id) {
            const emp = employees.find(e => e.id === id);
            if (!emp) return;
            const name = document.getElementById('editName').value.trim();
            const email = document.getElementById('editEmail').value.trim();
            const dept = document.getElementById('editDept').value.trim();
            const desig = document.getElementById('editDesig').value.trim();
            const salary = parseInt(document.getElementById('editSalary').value);
            if (name && email && dept && desig && salary > 0) {
                emp.name = name;
                emp.email = email;
                emp.department = dept;
                emp.designation = desig;
                emp.salary = salary;
                const basic = salary * 0.5;
                const hra = salary * 0.2;
                const da = salary * 0.1;
                const bonus = salary * 0.05;
                const deductions = salary * 0.12;
                payrollData[emp.id] = {
                    basic: Math.round(basic),
                    hra: Math.round(hra),
                    da: Math.round(da),
                    bonus: Math.round(bonus),
                    deductions: Math.round(deductions),
                    net: Math.round(basic + hra + da + bonus - deductions)
                };
                closeModal();
                renderAll();
                showToast('Employee updated');
            } else {
                showToast('Please fill all fields');
            }
        }

        function closeModal() {
            document.getElementById('modal').classList.remove('open');
        }

        // ==================== REGISTER ====================
        function registerEmployee(e) {
            e.preventDefault();
            const name = document.getElementById('regName').value.trim();
            const email = document.getElementById('regEmail').value.trim();
            const dept = document.getElementById('regDepartment').value;
            const desig = document.getElementById('regDesignation').value;
            const salary = parseInt(document.getElementById('regSalary').value);
            const status = document.getElementById('regStatus').value;

            if (!name || !email || !salary || salary < 1000) {
                showToast('Please fill all required fields');
                return;
            }

            const newEmp = {
                id: employeeIdCounter++,
                name,
                email,
                department: dept,
                designation: desig,
                status,
                salary
            };
            employees.push(newEmp);

            attendance[newEmp.id] = { present: 0, total: 22, today: 'Absent' };

            const basic = salary * 0.5;
            const hra = salary * 0.2;
            const da = salary * 0.1;
            const bonus = salary * 0.05;
            const deductions = salary * 0.12;
            payrollData[newEmp.id] = {
                basic: Math.round(basic),
                hra: Math.round(hra),
                da: Math.round(da),
                bonus: Math.round(bonus),
                deductions: Math.round(deductions),
                net: Math.round(basic + hra + da + bonus - deductions)
            };

            document.getElementById('registerForm').reset();
            renderAll();
            showToast('Employee registered successfully!');
            navigateTo('employees');
        }

        // ==================== ATTENDANCE ====================
        function markAttendance() {
            employees.forEach(emp => {
                if (attendance[emp.id]) {
                    attendance[emp.id].today = 'Present';
                    attendance[emp.id].present = (attendance[emp.id].present || 0) + 1;
                }
            });
            renderAll();
            showToast('All marked present');
        }

        function markLeave() {
            const modal = document.getElementById('modal');
            const content = document.getElementById('modalContent');
            content.innerHTML = `
                <h2><i class="fas fa-clock"></i> Mark Leave</h2>
                <div class="form-group">
                    <label>Select Employee</label>

                </div>

                    <button class="btn" onclick="closeModal()">Cancel</button>

                </div>
            `;
            modal.classList.add('open');
        }

        function confirmLeave() {
            const id = parseInt(document.getElementById('leaveEmployee').value);
            const emp = employees.find(e => e.id === id);
            if (emp) {
                emp.status = 'On Leave';
                if (attendance[id]) {
                    attendance[id].today = 'Absent';
                }
                closeModal();
                renderAll();
                showToast(`${emp.name} marked on leave`);
            }
        }

        function calculatePayroll() {

                const salary = emp.salary;
                const basic = salary * 0.5;
                const hra = salary * 0.2;
                const da = salary * 0.1;
                const bonus = salary * 0.05;
                const deductions = salary * 0.12;
                payrollData[emp.id] = {
                    basic: Math.round(basic),
                    hra: Math.round(hra),
                    da: Math.round(da),
                    bonus: Math.round(bonus),
                    deductions: Math.round(deductions),
                    net: Math.round(basic + hra + da + bonus - deductions)
                };
            });
            renderAll();
            showToast('Payroll calculated!');
        }
